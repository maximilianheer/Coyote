cmake_minimum_required(VERSION 3.21)
project(hbm_transport LANGUAGES CXX HIP)
set(CMAKE_CXX_STANDARD 17)
set(EN_GPU 1 CACHE STRING "GPU DMA-BUF support" FORCE)
set(CMAKE_HIP_ARCHITECTURES gfx90a)
find_package(pybind11 CONFIG REQUIRED)
add_subdirectory(${COYOTE_SW} coyote)
pybind11_add_module(hbm_transport transport.cpp)
set_source_files_properties(transport.cpp PROPERTIES LANGUAGE HIP)
target_include_directories(hbm_transport PRIVATE ../../gpu)
target_link_libraries(hbm_transport PRIVATE Coyote)
