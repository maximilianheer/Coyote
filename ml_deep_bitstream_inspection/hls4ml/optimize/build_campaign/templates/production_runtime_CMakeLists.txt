cmake_minimum_required(VERSION 3.16)
project(ProductionRuntime LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(EN_GPU 0 CACHE STRING "GPU disabled" FORCE)
add_subdirectory("${COYOTE_SW}" coyote)
add_library(CoyoteInference SHARED host_libs.cpp)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/production_host_csr.cpp")
    target_sources(CoyoteInference PRIVATE production_host_csr.cpp)
endif()
target_include_directories(CoyoteInference PRIVATE "${COYOTE_SW}/include/coyote")
target_link_libraries(CoyoteInference PRIVATE Coyote)
set_target_properties(CoyoteInference PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
