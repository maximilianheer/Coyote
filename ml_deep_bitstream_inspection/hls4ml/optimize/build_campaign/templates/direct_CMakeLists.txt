cmake_minimum_required(VERSION 3.16)
project(direct_bench LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
option(DIRECT_GPU "Enable GPU P2P runner (requires original baseline adapter)" OFF)
set(EN_GPU ${DIRECT_GPU} CACHE BOOL "Coyote GPU support" FORCE)
add_subdirectory("${COYOTE_SW}" coyote)
add_executable(direct_bench direct_host.cpp)
target_link_libraries(direct_bench PRIVATE Coyote ${CMAKE_DL_LIBS})
if(DIRECT_GPU)
    target_compile_definitions(direct_bench PRIVATE DIRECT_GPU=1)
endif()
