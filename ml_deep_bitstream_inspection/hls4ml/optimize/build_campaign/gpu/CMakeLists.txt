cmake_minimum_required(VERSION 3.21)
project(gpu_transport LANGUAGES CXX HIP)
set(CMAKE_CXX_STANDARD 17)
set(EN_GPU 1 CACHE STRING "GPU support" FORCE)
set(HIP_PATH /opt/rocm CACHE PATH "HIP")
set(CMAKE_HIP_ARCHITECTURES gfx90a)
add_subdirectory(${COYOTE_SW} coyote)
add_executable(p2p_probe p2p_probe.cpp)
set_source_files_properties(p2p_probe.cpp PROPERTIES LANGUAGE HIP)
target_link_libraries(p2p_probe PRIVATE Coyote)
add_executable(mailbox_probe mailbox_probe.cpp)
set_source_files_properties(mailbox_probe.cpp PROPERTIES LANGUAGE HIP)
target_link_libraries(mailbox_probe PRIVATE Coyote)
if(DEFINED pybind11_DIR)
find_package(pybind11 CONFIG REQUIRED)
pybind11_add_module(gpu_transport transport.cpp)
set_source_files_properties(transport.cpp PROPERTIES LANGUAGE HIP)
target_link_libraries(gpu_transport PRIVATE Coyote)
endif()
